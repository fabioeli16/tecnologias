1-ELIMINAR VERSION ANTERIOR DEL PATH
#PRIMERO VALIDAR
[Environment]::GetEnvironmentVariable("Path", "User")


#GUARDAR PATH EN VARIABLE TEMPORAL
$path = [Environment]::GetEnvironmentVariable("Path", "User")

#ACTUALIZAR PATH
$path = $path -replace ";C:\\nodev22\\npm-global", ""
$path = $path -replace ";C:\\nodev22", ""

[Environment]::SetEnvironmentVariable("Path", $path, "User")


#VALIDAR NUEVAMENTE
[Environment]::GetEnvironmentVariable("Path", "User")
